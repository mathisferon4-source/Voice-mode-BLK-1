# VOICE FRANCE RPG

Application Android Kotlin + Jetpack Compose. Version de départ : 0.1.0.

## Fonctionnalités incluses
- Interface sombre en français.
- Navigation Accueil, Voix, Sons et Réglages.
- Demande de permission microphone.
- Écoute locale du microphone via AudioRecord et AudioTrack.
- Réglage du gain et de la vitesse de lecture.
- Sélection d'un fichier audio depuis le téléphone.
- Workflow GitHub Actions pour compiler un APK Debug.

## Limites de cette version
Cette version est un prototype de départ. Elle ne propose pas encore un véritable changement indépendant de hauteur, un enregistrement audio persistant, une bibliothèque soundboard complète, ni un micro virtuel utilisable dans Discord, les jeux ou les appels. Android ne permet pas à une application ordinaire de remplacer globalement le microphone des autres applications. Le réglage de vitesse change également la hauteur et la durée. Utiliser un casque pour éviter le retour acoustique.

## Compilation
Avec Java 17 et Gradle 8.11.1 :

```bash
gradle --no-daemon :app:assembleDebug
```

APK : `app/build/outputs/apk/debug/app-debug.apk`.

## GitHub Actions
Pousser le projet sur la branche `main`, puis ouvrir l'onglet Actions. Télécharger l'artefact `Voice-France-RPG-APK` depuis une exécution réussie.

@Composable
private fun VoicePage(modifier: Modifier = Modifier) {
    val context = LocalContext.current
    val engine = remember { AudioEngine() }
    var running by remember { mutableStateOf(false) }
    var gain by remember { mutableFloatStateOf(1f) }
    var rate by remember { mutableFloatStateOf(1f) }
    var message by remember { mutableStateOf("Micro arrêté") }
    val permissionLauncher = rememberLauncherForActivityResult(ActivityResultContracts.RequestPermission()) { granted ->
        if (granted) {
            running = engine.start(gain, rate)
            message = if (running) "Écoute locale activée" else "Impossible de démarrer le micro"
        } else message = "Autorisation du micro refusée"
    }
    DisposableEffect(Unit) { onDispose { engine.stop() } }

    PageColumn(modifier) {
        Text("🎙️ Studio vocal", style = MaterialTheme.typography.headlineMedium, fontWeight = FontWeight.Bold)
        Card(colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surface)) {
            Column(Modifier.padding(18.dp), verticalArrangement = Arrangement.spacedBy(12.dp)) {
                Text(if (running) "🟢 Micro actif" else "⚫ Micro inactif", style = MaterialTheme.typography.titleLarge)
                Text(message)
                Button(
                    onClick = {
                        if (running) {
                            engine.stop(); running = false; message = "Micro arrêté"
                        } else if (ContextCompat.checkSelfPermission(context, Manifest.permission.RECORD_AUDIO) == PackageManager.PERMISSION_GRANTED) {
                            running = engine.start(gain, rate)
                            message = if (running) "Écoute locale activée" else "Impossible de démarrer le micro"
                        } else permissionLauncher.launch(Manifest.permission.RECORD_AUDIO)
                    }, modifier = Modifier.fillMaxWidth()
                ) { Text(if (running) "⏹ Arrêter" else "▶ Démarrer l'écoute") }
            }
        }
        Text("🔊 Volume de sortie : ${(gain * 100).toInt()} %")
        Slider(value = gain, onValueChange = { gain = it; engine.update(gain, rate) }, valueRange = 0f..2f)
        Text("⏩ Vitesse : ${(rate * 100).toInt()} %")
        Slider(value = rate, onValueChange = { rate = it; engine.update(gain, rate) }, valueRange = 0.75f..1.35f)
        Text("La vitesse modifie aussi la hauteur et la durée du son. Ce réglage n'est pas un vrai réglage indépendant de tonalité.")
        Text("⚠️ Cette version renvoie le son du micro vers le téléphone. Elle ne crée pas de micro virtuel pour Discord, les jeux ou les appels. Utilise un casque pour limiter l'écho.", color = MaterialTheme.colorScheme.secondary)
    }
}

package fr.voicefrancerpg.app.audio

import android.media.AudioAttributes
import android.media.AudioFormat
import android.media.AudioRecord
import android.media.AudioTrack
import android.media.MediaRecorder
import android.media.audiofx.AcousticEchoCanceler
import android.media.audiofx.NoiseSuppressor
import android.os.Process
import kotlin.math.max

/** Monitoring local du micro. Ce moteur ne crée pas de micro virtuel Android. */
class AudioEngine {
    @Volatile private var running = false
    @Volatile private var outputGain = 1.0f
    @Volatile private var playbackRate = 1.0f
    private var worker: Thread? = null
    private var recorder: AudioRecord? = null
    private var track: AudioTrack? = null
    private var echoCanceler: AcousticEchoCanceler? = null
    private var noiseSuppressor: NoiseSuppressor? = null

    fun update(gain: Float, rate: Float) {
        outputGain = gain.coerceIn(0f, 2f)
        playbackRate = rate.coerceIn(0.75f, 1.35f)
        runCatching { track?.playbackRate = (SAMPLE_RATE * playbackRate).toInt() }
    }

    @Synchronized
    fun start(gain: Float, rate: Float): Boolean {
        if (running) return true
        update(gain, rate)
        var input: AudioRecord? = null
        var output: AudioTrack? = null
        return try {
            val minIn = AudioRecord.getMinBufferSize(SAMPLE_RATE, IN_CHANNEL, ENCODING)
            val minOut = AudioTrack.getMinBufferSize(SAMPLE_RATE, OUT_CHANNEL, ENCODING)
            if (minIn <= 0 || minOut <= 0) return false
            val bufferSize = max(minIn, minOut) * 2
            input = AudioRecord(
                MediaRecorder.AudioSource.MIC, SAMPLE_RATE,
                IN_CHANNEL, ENCODING, bufferSize
            )
            if (input.state != AudioRecord.STATE_INITIALIZED) return false

            output = AudioTrack.Builder()
                .setAudioAttributes(
                    AudioAttributes.Builder()
                        .setUsage(AudioAttributes.USAGE_MEDIA)
                        .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
                        .build()
                )
                .setAudioFormat(
                    AudioFormat.Builder()
                        .setEncoding(ENCODING)
                        .setSampleRate(SAMPLE_RATE)
                        .setChannelMask(OUT_CHANNEL)
                        .build()
                )
                .setBufferSizeInBytes(bufferSize)
                .setTransferMode(AudioTrack.MODE_STREAM)
                .setPerformanceMode(AudioTrack.PERFORMANCE_MODE_LOW_LATENCY)
                .build()
            if (output.state != AudioTrack.STATE_INITIALIZED) return false
            output.playbackRate = (SAMPLE_RATE * playbackRate).toInt()

            recorder = input
            track = output
            if (AcousticEchoCanceler.isAvailable()) {
                echoCanceler = AcousticEchoCanceler.create(input.audioSessionId)?.apply { enabled = true }
            }
            if (NoiseSuppressor.isAvailable()) {
                noiseSuppressor = NoiseSuppressor.create(input.audioSessionId)?.apply { enabled = true }
            }
            running = true
            input.startRecording()
            output.play()
            worker = Thread({ audioLoop(input, output) }, "VoiceFranceAudio").also { it.start() }
            true
        } catch (_: Exception) {
            running = false
            runCatching { input?.release() }
            runCatching { output?.release() }
            recorder = null
            track = null
            false
        }
    }

    private fun audioLoop(input: AudioRecord, output: AudioTrack) {
        Process.setThreadPriority(Process.THREAD_PRIORITY_AUDIO)
        val buffer = ShortArray(1024)
        while (running) {
            val count = input.read(buffer, 0, buffer.size, AudioRecord.READ_BLOCKING)
            if (count > 0) {
                val gain = outputGain
                for (i in 0 until count) {
                    buffer[i] = (buffer[i].toFloat() * gain)
                        .toInt().coerceIn(Short.MIN_VALUE.toInt(), Short.MAX_VALUE.toInt()).toShort()
                }
                var offset = 0
                while (offset < count && running) {
                    val written = output.write(buffer, offset, count - offset, AudioTrack.WRITE_BLOCKING)
                    if (written <= 0) break
                    offset += written
                }
            } else if (count < 0) break
        }
    }

    @Synchronized
    fun stop() {
        running = false
        runCatching { recorder?.stop() }
        runCatching { worker?.join(400) }
        worker = null
        runCatching { track?.pause(); track?.flush(); track?.stop() }
        runCatching { echoCanceler?.release() }
        runCatching { noiseSuppressor?.release() }
        runCatching { recorder?.release() }
        runCatching { track?.release() }
        recorder = null
        track = null
        echoCanceler = null
        noiseSuppressor = null
    }

    companion object {
        private const val SAMPLE_RATE = 24000
        private const val ENCODING = AudioFormat.ENCODING_PCM_16BIT
        private const val IN_CHANNEL = AudioFormat.CHANNEL_IN_MONO
        private const val OUT_CHANNEL = AudioFormat.CHANNEL_OUT_MONO
    }
}
